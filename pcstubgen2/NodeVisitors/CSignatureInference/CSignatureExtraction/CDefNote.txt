typedef struct PyMethodDef {
    const char  *ml_name;   /* Python 中的方法名 */
    PyCFunction  ml_meth;   /* 对应的 C 函数指针 */
    int          ml_flags;  /* 调用约定 */
    const char  *ml_doc;    /* 文档字符串 */
} PyMethodDef;

ml = method list

# PyCFunction定义
typedef PyObject* (*PyCFunction)(PyObject* self, PyObject* args);

# PyCFunctionWithKeywords定义
typedef PyObject *(*PyCFunctionWithKeywords)(
    PyObject *self,
    PyObject *args,
    PyObject *kwargs
);

# PyCMethod定义
typedef PyObject *(*PyCMethod)(
    PyObject *self,
    PyTypeObject *defining_class,
    PyObject *const *args,
    Py_ssize_t nargs,
    PyObject *kwnames
);
